cmake_minimum_required(VERSION 3.20)
project(OpenScale VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OPENSCALE_BUILD_TESTS "Build tests" ON)
option(OPENSCALE_ENABLE_CUDA "Enable CUDA backend" OFF)
option(OPENSCALE_ENABLE_HIP "Enable HIP backend" OFF)
option(OPENSCALE_ENABLE_METAL "Enable Metal backend" OFF)
option(OPENSCALE_ENABLE_VULKAN "Enable Vulkan backend" OFF)
add_library(openscale_core STATIC
 src/av_format.cpp src/import.cpp src/progress.cpp src/hardware.cpp src/feasibility.cpp src/runtime.cpp
 src/backends.cpp src/memory.cpp src/graph.cpp src/kv_cache.cpp src/tokenizer.cpp src/model.cpp src/planner.cpp src/modules.cpp src/benchmark.cpp)
target_include_directories(openscale_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(openscale_core PUBLIC OPENSCALE_API_LEVEL=1)
if(OPENSCALE_ENABLE_CUDA)
 target_compile_definitions(openscale_core PUBLIC OPENSCALE_ENABLE_CUDA=1)
endif()
if(OPENSCALE_ENABLE_HIP)
 target_compile_definitions(openscale_core PUBLIC OPENSCALE_ENABLE_HIP=1)
endif()
if(OPENSCALE_ENABLE_METAL)
 target_compile_definitions(openscale_core PUBLIC OPENSCALE_ENABLE_METAL=1)
endif()
if(OPENSCALE_ENABLE_VULKAN)
 target_compile_definitions(openscale_core PUBLIC OPENSCALE_ENABLE_VULKAN=1)
endif()
if(MSVC)
 target_compile_options(openscale_core PRIVATE /O2 /W4)
else()
 target_compile_options(openscale_core PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
add_executable(openscale tools/openscale.cpp)
target_link_libraries(openscale PRIVATE openscale_core)
if(OPENSCALE_BUILD_TESTS)
 enable_testing()
 add_executable(openscale_tests tests/test_core.cpp)
 target_link_libraries(openscale_tests PRIVATE openscale_core)
 add_test(NAME openscale_core COMMAND openscale_tests)
endif()

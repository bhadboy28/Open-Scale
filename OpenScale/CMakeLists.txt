cmake_minimum_required(VERSION 3.20)
project(OpenScale LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(openscale_core src/core.cpp src/av.cpp src/memory.cpp src/math.cpp src/modules.cpp src/quantize.cpp src/api.cpp)
add_library(openscale_shared SHARED src/core.cpp src/av.cpp src/memory.cpp src/math.cpp src/modules.cpp src/quantize.cpp src/api.cpp)
target_include_directories(openscale_core PUBLIC include)
target_include_directories(openscale_shared PUBLIC include)
if(MSVC)
 target_compile_options(openscale_core PRIVATE /O2 /arch:AVX2)
 target_compile_options(openscale_shared PRIVATE /O2 /arch:AVX2)
else()
 target_compile_options(openscale_core PRIVATE -O3 -ffast-math -mavx2 -mfma -march=native)
 target_compile_options(openscale_shared PRIVATE -O3 -ffast-math -mavx2 -mfma -march=native)
endif()
add_executable(openscale tools/openscale.cpp)
target_link_libraries(openscale PRIVATE openscale_core)
enable_testing()
add_executable(openscale_tests tests/test_core.cpp)
target_link_libraries(openscale_tests PRIVATE openscale_core)
add_test(NAME openscale_tests COMMAND openscale_tests)
